const ALLOW = [
  'coderacrop@gmail.com'
  // เพิ่มอีเมลทีมที่นี่ เช่น 'name@gmail.com',
];
const TRACK = '1L6uNcsjpeUFyWU095KSMtzMCJS0pFjpaYQ_hebUklJY';
const READ_OK = [TRACK, '1e8jFrTSBHjAEUcIPINBBW1UE7r6pkukEn3q1ztauvpk', '1V6OU5GomJK-gguxfR2toUQs6kgykOA50w5e-P1Qm-Zs', '1AEnp81GszEhYIMoVbuB_a_-xGq053RfjPtZssKprV8M'];
const WRITE_OK = [TRACK];
const LS_TAB = 'linestudio';

function me_() { return String(Session.getActiveUser().getEmail() || '').toLowerCase(); }
function ok_() { const m = me_(); return ALLOW.map(s => s.toLowerCase()).indexOf(m) >= 0; }
function guard_() { if (!ok_()) throw new Error('บัญชีนี้ไม่มีสิทธิ์ใช้ศูนย์งาน'); }

function doGet() {
  if (!ok_()) {
    const m = me_() || 'นี้';
    return HtmlService.createHtmlOutput('<div style="font-family:system-ui,sans-serif;max-width:480px;margin:80px auto;padding:0 16px;color:#111"><h2 style="margin:0 0 8px">Codera ศูนย์งาน</h2><p style="color:#55554f">บัญชี ' + m.replace(/[<>&]/g, '') + ' ยังไม่มีสิทธิ์เข้า · ส่งอีเมลนี้ให้คิมเพื่อเพิ่มสิทธิ์</p></div>')
      .setTitle('Codera ศูนย์งาน').addMetaTag('viewport', 'width=device-width, initial-scale=1');
  }
  return HtmlService.createHtmlOutputFromFile('hub').setTitle('Codera ศูนย์งาน')
    .addMetaTag('viewport', 'width=device-width, initial-scale=1');
}

function trim_(v) {
  let n = v.length; while (n && v[n - 1].every(c => c === '')) n--;
  return v.slice(0, n).map(r => { let k = r.length; while (k && r[k - 1] === '') k--; return r.slice(0, k); });
}

function getValues(id, range) {
  guard_(); if (READ_OK.indexOf(id) < 0) throw new Error('ชีทนี้ไม่อยู่ในรายการ');
  const ss = SpreadsheetApp.openById(id);
  const sh = ss.getSheetByName(String(range).split('!')[0].replace(/^'|'$/g, ''));
  if (!sh) return [];
  return trim_(ss.getRange(range).getDisplayValues());
}

function updateValues(id, range, values) {
  guard_(); if (WRITE_OK.indexOf(id) < 0) throw new Error('เขียนชีทนี้ไม่ได้');
  const w = Math.max.apply(null, values.map(r => r.length));
  const rows = values.map(r => { const x = r.slice(); while (x.length < w) x.push(''); return x; });
  const start = SpreadsheetApp.openById(id).getRange(range);
  const rg = start.getSheet().getRange(start.getRow(), start.getColumn(), rows.length, w);
  rg.setValues(rows);
  return { updatedRange: rg.getA1Notation(), updatedRows: rows.length };
}

function listFolder(id) {
  guard_(); if (!id) return [];
  const f = DriveApp.getFolderById(id), out = [];
  const fo = f.getFolders(); while (fo.hasNext()) { const x = fo.next(); out.push({ id: x.getId(), title: x.getName(), mimeType: 'application/vnd.google-apps.folder', modifiedTime: x.getLastUpdated().toISOString(), viewUrl: x.getUrl() }); }
  const fi = f.getFiles(); while (fi.hasNext()) { const x = fi.next(); out.push({ id: x.getId(), title: x.getName(), mimeType: x.getMimeType(), modifiedTime: x.getLastUpdated().toISOString(), viewUrl: x.getUrl() }); }
  return out;
}

/* LINE Studio · เก็บแผนในแท็บ linestudio ของชีทงาน (id · เวลา · JSON แบ่งช่องละ 45,000 ตัวอักษร) */
function lsSheet_() {
  const ss = SpreadsheetApp.openById(TRACK);
  let sh = ss.getSheetByName(LS_TAB);
  if (!sh) { sh = ss.insertSheet(LS_TAB); sh.getRange(1, 1, 1, 3).setValues([['id', 'updatedAt', 'json']]); sh.hideSheet(); }
  return sh;
}
function lsList() {
  guard_(); const sh = lsSheet_(), n = sh.getLastRow(); if (n < 2) return [];
  return sh.getRange(2, 1, n - 1, sh.getLastColumn()).getValues().filter(r => r[0]).map(r => { try { return JSON.parse(r.slice(2).join('')); } catch (e) { return null; } }).filter(Boolean);
}
function lsSet(id, json) {
  guard_(); const sh = lsSheet_(), parts = [];
  for (let i = 0; i < json.length; i += 45000) parts.push(json.slice(i, i + 45000));
  const ids = sh.getLastRow() > 1 ? sh.getRange(2, 1, sh.getLastRow() - 1, 1).getValues().map(r => String(r[0])) : [];
  let row = ids.indexOf(String(id)); row = row < 0 ? sh.getLastRow() + 1 : row + 2;
  const lc = Math.max(sh.getLastColumn(), 3); sh.getRange(row, 1, 1, lc).clearContent();
  sh.getRange(row, 1, 1, 2 + parts.length).setValues([[String(id), Date.now()].concat(parts)]);
  return true;
}
function lsDel(id) {
  guard_(); const sh = lsSheet_(); if (sh.getLastRow() < 2) return true;
  const ids = sh.getRange(2, 1, sh.getLastRow() - 1, 1).getValues().map(r => String(r[0]));
  const i = ids.indexOf(String(id)); if (i >= 0) sh.deleteRow(i + 2);
  return true;
}
