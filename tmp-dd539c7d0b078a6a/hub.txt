function barsSVG(labels,vals,color){

async function loadFolder(){
  const fl=$('#flist'), fc=$('#fcrumbs'); if(!fl)return; if(!BV.stack.length){fl.innerHTML='<div class="empty">ยังไม่มีโฟลเดอร์ผูกกับแบรนด์นี้</div>';fc.innerHTML='';return;}
  const cur=BV.stack[BV.stack.length-1];
  fc.innerHTML=BV.stack.map((s,i)=>i===BV.stack.length-1?'<span class="cur">'+esc(s.n)+'</span>':'<button type="button" data-up="'+i+'">'+esc(s.n)+'</button><span class="sep">›</span>').join('')+'<a class="btn sm" style="margin-left:auto" href="'+FD(cur.id)+'" target="_blank" rel="noopener">'+ICON.ext+'เปิดใน Drive</a>';
  fc.querySelectorAll('[data-up]').forEach(b=>b.onclick=()=>{BV.stack=BV.stack.slice(0,+b.dataset.up+1);loadFolder();});
  if(!S.mcp){fl.innerHTML='<a class="file" href="'+FD(cur.id)+'" target="_blank" rel="noopener">'+ICON.folder+'<span class="nm">เปิด '+esc(cur.n)+' ใน Drive</span></a>';return;}
  if(BV.cache[cur.id]){drawFiles(BV.cache[cur.id]);return;}
  fl.innerHTML='<div class="skel"></div><div class="skel"></div>';
  try{ const res=await S.mcp.callTool(DR,'search_files',{query:"parentId = '"+cur.id+"' and trashed = false",excludeContentSnippets:true,pageSize:100},{cache:{staleTime:60000}});
    const files=((res.payload||{}).files||[]).slice(); files.sort((a,b)=>{const fa=(a.mimeType||'').includes('folder'),fb=(b.mimeType||'').includes('folder');return fa!==fb?(fa?-1:1):String(a.title).localeCompare(String(b.title),'th');});
    BV.cache[cur.id]=files; if(BV.stack[BV.stack.length-1].id===cur.id)drawFiles(files);
  }catch(e){ const code=e&&e.code; fl.innerHTML='<div class="empty">'+esc(code==='not_granted'?'ยังไม่ได้อนุญาตให้หน้านี้ใช้ Google Drive':code==='server_not_connected'?'บัญชีนี้ยังไม่ได้เชื่อม Google Drive':'อ่านโฟลเดอร์ไม่สำเร็จ')+'<br><a class="btn sm" style="margin-top:8px" href="'+FD(cur.id)+'" target="_blank" rel="noopener">เปิดใน Drive แทน</a></div>'; }
}
function drawFiles(files){ const fl=$('#flist'); if(!fl)return; if(!files.length){fl.innerHTML='<div class="empty">โฟลเดอร์นี้ยังว่าง</div>';return;}
  fl.innerHTML=files.map((f,i)=>{ const isF=(f.mimeType||'').includes('folder'), dt=f.modifiedTime?new Date(f.modifiedTime).toLocaleDateString('th-TH',{day:'numeric',month:'short'}):'';
    return isF?'<button type="button" class="file" data-f="'+i+'">'+ICON.folder+'<span class="nm">'+esc(f.title)+'</span><span class="dt">'+esc(dt)+'</span></button>':'<a class="file" href="'+esc(f.viewUrl||('https://drive.google.com/open?id='+f.id))+'" target="_blank" rel="noopener">'+iconFor(f.mimeType)+'<span class="nm">'+esc(f.title)+'</span><span class="dt">'+esc(dt)+'</span></a>'; }).join('');
  fl.querySelectorAll('[data-f]').forEach(b=>b.onclick=()=>{const f=files[+b.dataset.f];BV.stack.push({n:f.title,id:f.id});loadFolder();}); }
